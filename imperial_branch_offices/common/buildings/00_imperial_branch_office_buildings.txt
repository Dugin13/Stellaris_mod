# Custom Branch Office Buildings for Regular Empires

# Embassy Complex - Provides diplomatic weight and amenities
building_imperial_embassy = {
	base_buildtime = 360
	owner_type = corporate
	
	category = government
	
	potential = {
		exists = owner
		owner = { 
			OR = {
				has_country_flag = imperial_branch_offices_enabled
				has_ascension_perk = ap_diplomatic_expansion
			}
		}
		has_branch_office = yes
	}
	
	allow = {
		num_buildings = { type = building_imperial_embassy value = 0 }
	}
	
	resources = {
		category = planet_branch_office_buildings
		cost = {
			minerals = 500
		}
		upkeep = {
			energy = 2
		}
	}
	
	planet_modifier = {
		planet_amenities_add = 5
	}
	
	country_modifier = {
		diplo_weight_mult = 0.03
	}
	
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = { is_gestalt = no }
		}
		modifier = {
			job_clerk_add = 2
		}
	}
	
	ai_weight = {
		weight = 50
	}
}
