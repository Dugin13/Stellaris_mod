# Wilderness Ring World Districts
# Wilderness-specific ring world districts that appear on specialized segments
# Based on Ultimate Technologies district structure

# Wilderness City District (Housing Focus)
district_rw_wilderness_city = {
	base_buildtime = 240
	is_capped_by_modifier = no
	
	show_on_uncolonized = {
		is_planet_class = pc_ringworld_city
		from = { is_wilderness_empire = yes }
	}
	
	potential = {
		is_planet_class = pc_ringworld_city
		exists = owner
		owner = { is_wilderness_empire = yes }
	}
	
	allow = {
		owner = { has_ascension_perk = ap_ring_world_pioneers }
	}
	
	conversion_ratio = 1
	convert_to = {
		district_rw_city
		district_rw_hive
	}
	
	resources = {
		category = planet_districts
		cost = {
			minerals = 1000
			exotic_gases = 50
		}
		upkeep = {
			energy = 5
			exotic_gases = 1
		}
	}
	
	planet_modifier = {
		planet_housing_add = 50
		planet_amenities_add = 20
	}
	
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				has_valid_civic = civic_agrarian_idyll
			}
		}
		modifier = {
			planet_housing_add = -10
		}
	}
	
	triggered_desc = {
		trigger = {
			exists = owner
			owner = {
				is_gestalt = yes
			}
		}
		text = job_maintenance_drone_effect_desc
	}
	
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				is_gestalt = yes
			}
		}
		modifier = {
			job_maintenance_drone_add = 10
		}
	}
}
