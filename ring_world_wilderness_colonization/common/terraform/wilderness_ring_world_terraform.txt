# Wilderness-compatible terraform links for ring worlds and relic worlds
# These allow "terraform to self" so wilderness empires can colonize them
# Segment specialization is handled through decisions (see common/decisions)

# Ring World Habitable - terraform to itself (for colonization)
terraform_link = {
	from = "pc_ringworld_habitable"
	to = "pc_ringworld_habitable"
	
	resources = {
		category = terraforming
		cost = {
			energy = 0
		}
	}
	
	duration = 1
	
	potential = {
		exists = owner
		owner = {
			is_wilderness_empire = yes
			has_ascension_perk = ap_ring_world_pioneers
		}
	}
	
	condition = {
		has_ascension_perk = ap_ring_world_pioneers
	}
	
	ai_weight = {
		weight = 100
	}
}
