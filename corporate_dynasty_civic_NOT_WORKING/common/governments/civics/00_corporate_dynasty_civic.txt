# Corporate Dynasty Civic
# Enables hereditary succession for Corporate authority

civic_corporate_dynasty = {
	potential = {
		authority = { value = auth_corporate }
	}
	
	possible = {
		ethics = {
			NOT = { value = ethic_egalitarian }
			NOT = { value = ethic_fanatic_egalitarian }
		}
		authority = { value = auth_corporate }
	}
	
	# Cannot be combined with civics that conflict with hereditary rule
	random_weight = { base = 5 }
	
	modifier = {
		ruler_skill_levels = 1
		country_unity_produces_mult = 0.05
	}
	
	# Key attributes for hereditary succession
	has_heir = yes
	can_have_emergency_elections = no
	election_type = none
	
	description = "civic_corporate_dynasty_effects"
	
	ai_weight = {
		base = 5
		modifier = {
			factor = 2
			has_ethic = ethic_authoritarian
		}
		modifier = {
			factor = 3
			has_ethic = ethic_fanatic_authoritarian
		}
		modifier = {
			factor = 0
			OR = {
				has_ethic = ethic_egalitarian
				has_ethic = ethic_fanatic_egalitarian
			}
		}
	}
}
